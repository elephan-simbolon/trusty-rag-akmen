       def embed_batch(texts: list[str], is_query: bool = False) -> list[list[float]]:

               texts = [settings.embedding_query_instruction + t for t in texts]

    7. Create `rerank(query: str, documents: list[str], top_k: int = 5) -> list[dict]` — reranking via SiliconFlow:
       Uses the reranker model endpoint. Returns list of `{"index": int, "score": float, "text": str}` sorted by score descending:
       ```python
       @retry(...)  # same retry config
       def rerank(query: str, documents: list[str], top_k: int = 5) -> list[dict]:
           client = get_openai_client()
           # SiliconFlow reranker uses a specific endpoint format
           import httpx
           response = httpx.post(
               f"{settings.siliconflow_base_url}/rerank",
               headers={"Authorization": f"Bearer {settings.siliconflow_api_key.get_secret_value()}"},
               json={
                   "model": settings.reranker_model,
                   "query": query,
                   "documents": documents,
                   "top_n": top_k,
               },
               timeout=60.0
           )
           response.raise_for_status()
           results = response.json()["results"]
           return [
               {"index": r["index"], "score": r["relevance_score"], "text": documents[r["index"]]}
               for r in sorted(results, key=lambda x: x["relevance_score"], reverse=True)
           ]
       ```
