   ```python
   def render_query_type_badge(query_type: str | None) -> None:
       """Render inline badge showing query classification type.
       Simple queries show no badge (default path, no visual noise).
       """
       if not query_type or query_type == "Simple":
           return

       badge_map = {
           "Calculation": ("Kalkulasi", "query-type-badge query-type-badge--calc"),
           "Medium": ("Analisis", "query-type-badge query-type-badge--medium"),
           "Complex": ("Mendalam", "query-type-badge query-type-badge--complex"),
       }

       label, css_class = badge_map.get(query_type, (query_type, "query-type-badge query-type-badge--medium"))
       st.markdown(
           f'<span class="{css_class}">{label}</span>',
           unsafe_allow_html=True,
       )
   ```
