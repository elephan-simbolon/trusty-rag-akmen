   import re
   _CALC_KEYWORDS = frozenset([
       "hitung", "hitunglah", "berapa", "kalkulasi", "kalkulasikan",
       "bep", "break-even", "break even",
   ])
   _NUMBER_PATTERN = re.compile(r"\d[\d.,]*")

   def is_calculation_query(query: str) -> bool:
       q_lower = query.lower()
       has_keyword = any(kw in q_lower for kw in _CALC_KEYWORDS)
       has_number = bool(_NUMBER_PATTERN.search(query))
       return has_keyword and has_number
   ```

   def route_node(state: RAGState) -> dict:
       """Classify query type. Rule-based first (RETR-06), then default to Simple.

       For Phase 3 v1: Calculation detected by rule (numbers + keywords),
       everything else defaults to Simple. Full 4-tier LLM classifier
       deferred to Phase 4 if MON-01 monitoring reveals it's needed.

       Always resets crag_iterations to 0 and crag_grade to None to prevent
       MemorySaver persistence from prior turns (Pitfall 1 from RESEARCH.md).
       """
       query = state["query"]

       if is_calculation_query(query):
           return {
               "query_type": "Calculation",
               "llm_call_count": 0,
               "crag_iterations": 0,
               "crag_grade": None,
           }

       return {
           "query_type": "Simple",
           "llm_call_count": 0,
           "crag_iterations": 0,
           "crag_grade": None,
       }
   ```
