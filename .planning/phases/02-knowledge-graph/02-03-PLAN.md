   def generate_response(query: str, context_docs: list[dict], graph_context: str = "") -> dict:
       """
       Generate a bilingual response (Indonesian prose + English technical terms)
       with citations from retrieved context.

       When graph_context is provided (Phase 2+), uses SYSTEM_PROMPT_SYNTHESIS
       with explicit multi-source attribution instructions.
       When graph_context is empty (Phase 1 compatibility), uses SYSTEM_PROMPT_GENERATOR.

       Returns: dict with 'response' (str) and 'citations' (list[dict]).
       """
       glossary_snippet = _build_glossary_snippet()
       context_block = _build_context_block(context_docs)

       if graph_context:
           system_prompt = SYSTEM_PROMPT_SYNTHESIS.format(glossary_snippet=glossary_snippet)
           user_content = (
               f"Konteks dari knowledge graph:\n{graph_context}\n\n"
               f"Konteks dari textbook passages:\n{context_block}\n\n"
               f"Pertanyaan: {query}\n\n"
               "Instruksi: Sebutkan secara eksplisit sumber textbook (nama pengarang) "
               "untuk setiap klaim yang berbeda antara penulis."
           )
       else:
           system_prompt = SYSTEM_PROMPT_GENERATOR.format(glossary_snippet=glossary_snippet)
           user_content = f"Konteks dari textbook:\n\n{context_block}\n\nPertanyaan: {query}"

       messages = [
           {"role": "system", "content": system_prompt},
           {"role": "user", "content": user_content},
       ]

       response_text = generate(messages, temperature=0.3)
       citations = build_citations(context_docs)

       # Append citation block to response
       if citations:
           citation_lines = [f"- {c['formatted']}" for c in citations]
           citation_block = "\n\n**Sumber Referensi:**\n" + "\n".join(citation_lines)
           full_response = response_text + citation_block
       else:
           full_response = response_text

       return {
           "response": full_response,
           "citations": citations,
       }
   ```

   def generate_node(state: RAGState) -> dict:
       """Generate bilingual response with citations.
       Phase 2: merges graph_docs context into generation when available.
       """
       if state.get("error"):
           return {
               "response": f"Terjadi kesalahan: {state['error']}",
               "citations": [],
           }

       docs = state.get("reranked_docs") or state.get("retrieved_docs") or []
       if not docs:
           return {
               "response": "Tidak ditemukan referensi relevan untuk pertanyaan ini di korpus textbook yang tersedia.",
               "citations": [],
           }

       # Extract graph context text if available (Phase 2+)
       graph_context = ""
       graph_docs = state.get("graph_docs") or []
       if graph_docs:
           graph_context = "\n\n".join(doc["text"] for doc in graph_docs if doc.get("text"))

       try:
           result = generate_response(
               query=state["query"],
               context_docs=docs,
               graph_context=graph_context,
           )
           return {
               "response": result["response"],
               "citations": result["citations"],
           }
       except Exception as e:
           logger.error(f"Generation failed: {e}")
           return {
               "response": f"Gagal menghasilkan respons: {e}",
               "citations": [],
           }
   ```
